# C4 model

The diagrams use Mermaid's C4 syntax, which renders on GitHub and in most Markdown viewers. The [architecture document](../architecture.md) is the narrative companion.

## Level 1 — System context

Weft sits between the people who design and supervise agent graphs and the external systems agents act on. Model servers are external systems behind a provider abstraction. "Local" model servers (Ollama, vLLM, TGI) run in the same compose stack but are still separate systems.

```mermaid
C4Context
  title System context — Weft agent factory

  Person(designer, "Graph designer", "Writes graph documents, prompts and tool grants")
  Person(operator, "Operator / approver", "Starts and stops graphs, injects work, answers approvals")
  System_Ext(ci, "CI / external systems", "Inject work and read results through the API")

  System(weft, "Weft", "Runs agent graphs as durable loops with queues, permissions, HITL and observability")

  System_Ext(models, "Model servers", "Ollama, vLLM, TGI, OpenAI-compatible, Anthropic-compatible, OpenRouter")
  System_Ext(repos, "Source repositories", "Git remotes cloned into workspaces")
  System_Ext(web, "Web resources", "Fetched by http.fetch under host allowlists")
  System_Ext(idp, "OIDC identity provider", "Optional: issues JWTs for API users")

  Rel(designer, weft, "Edits graphs", "Console, CLI, YAML via HTTPS")
  Rel(operator, weft, "Operates, approves", "Console, CLI")
  Rel(ci, weft, "Injects messages, reads status", "REST/JSON")
  Rel(weft, models, "Chat completions with tools", "HTTP")
  Rel(weft, repos, "clone / worktrees", "git")
  Rel(weft, web, "Fetch", "HTTPS")
  Rel(weft, idp, "JWKS", "HTTPS")
```

## Level 2 — Containers

The compose stack in `docker-compose.yml`. `weftd` is the single control-plane binary and can be replicated. `agents` hosts the Python agents and holds no credentials. `sandbox-daemon` is a Docker daemon dedicated to Weft where docker-isolated agents' commands run ([sandbox runtime](../sandbox-runtime.md)).

```mermaid
C4Container
  title Containers — docker compose deployment

  Person(user, "Operator / designer")

  System_Boundary(lf, "Weft") {
    Container(console, "Web console", "React, TypeScript, xyflow", "Graph canvas, panels, approvals, timeline; served by weftd")
    Container(weftd, "weftd", "Go", "API, SSE, engine (reconciler, replicas, router), LLM and tool gateways, HITL, callback API")
    Container(agents, "Agent runtime", "Python, weft SDK", "Runtime Protocol v1 server: planner, researcher, developer, reviewer, qa; LangGraph/AutoGen/CrewAI adapters")
    ContainerQueue(nats, "Broker", "NATS JetStream (default); Redis Streams, RabbitMQ, Kafka or filesystem", "One stream per channel, one durable consumer per group")
    ContainerDb(pg, "Store", "PostgreSQL (default in compose); SQLite or filesystem", "Graph versions, records, events, history, executions, approvals, KV, leases")
    Container(ws, "Workspaces volume", "Filesystem + git", "Graph workspaces and per-replica worktrees")
    Container(sbx, "sandbox-daemon", "Docker-in-Docker", "One firewalled container per docker-isolated replica; images and caches in its own volume")
  }

  System_Boundary(obs, "Observability") {
    Container(otel, "OTel Collector", "otelcol-contrib", "OTLP in; span metrics; fan-out")
    ContainerDb(tempo, "Tempo", "Traces")
    ContainerDb(loki, "Loki", "Logs")
    ContainerDb(prom, "Prometheus", "Metrics + alert rules")
    Container(grafana, "Grafana", "Dashboards, Explore")
  }

  System_Ext(models, "Model servers", "Ollama / TGI profiles or remote APIs")

  Rel(user, console, "Uses", "HTTPS")
  Rel(console, weftd, "REST + SSE", "JSON")
  Rel(weftd, nats, "Publish / fetch / ack", "NATS")
  Rel(weftd, pg, "SQL")
  Rel(weftd, ws, "Files, git")
  Rel(weftd, sbx, "Create sandbox, docker exec", "Docker API, mutual TLS")
  Rel(sbx, ws, "Bind-mounts replica workspace")
  Rel(weftd, agents, "POST /v1/turn", "HTTP")
  Rel(agents, weftd, "Callbacks: tools, llm, emit, state", "HTTP + execution token")
  Rel(weftd, models, "Chat", "HTTP")
  Rel(weftd, otel, "Traces, logs", "OTLP/HTTP")
  Rel(prom, weftd, "Scrapes /metrics")
  Rel(otel, tempo, "Traces")
  Rel(otel, loki, "Logs")
  Rel(prom, otel, "Scrapes span metrics")
  Rel(grafana, prom, "Queries")
  Rel(grafana, tempo, "Queries")
  Rel(grafana, loki, "Queries")
```

## Level 3 — Components of `weftd`

```mermaid
C4Component
  title Components — weftd

  Container_Boundary(f, "weftd") {
    Component(api, "API server", "internal/api", "REST /api/v1, SSE stream, auth (tokens, OIDC), RBAC, middleware")
    Component(cb, "Callback API", "internal/api/internal.go", "/internal/v1/executions/{id}/*, per-execution token")
    Component(eng, "Engine / reconciler", "internal/engine", "Desired-state convergence, leases, autoscale, graph actions, status")
    Component(rep, "Replica loops", "engine/replica.go, turn.go", "Claim, turn, publish, ack/nack, stop handshake, ring")
    Component(router, "Graph model & router", "internal/graph", "Normalize, Validate, consumer groups, when-conditions, schedules")
    Component(rt, "Runtimes", "internal/runtime", "native (tool loop), external (Runtime Protocol), echo")
    Component(llmgw, "LLM gateway", "internal/llm", "Provider registry, retries, concurrency, metrics, gen_ai spans")
    Component(toolgw, "Tool gateway", "internal/tools", "Default-deny policy, constraints, rate limit, approvals, sandboxes, builtin tools")
    Component(hitl, "HITL service", "internal/hitl", "Approvals: create, wait, decide, expire, hooks")
    Component(bus, "Event bus", "internal/events", "Append-then-wake; store-tailing subscriptions")
    Component(wsm, "Workspace manager", "internal/workspace", "Graph dirs, git init/clone, worktrees")
    Component(obs, "Observability", "internal/observability", "Prometheus metrics, OTel tracer/logs, trace propagation in metadata")

  Rel(api, eng, "Graph CRUD, actions, inject, status")
  Rel(api, hitl, "Decide approvals")
  Rel(api, bus, "Subscribe (SSE)")
  Rel(cb, rep, "Resolve TurnEnv by execution token")
  Rel(eng, router, "Validate, topology, routing")
  Rel(eng, rep, "Start / stop / adopt")
  Rel(rep, rt, "Turn(TurnInput, TurnEnv)")
  Rel(rep, toolgw, "InvokeTool")
  Rel(rep, llmgw, "Chat")
  Rel(rep, hitl, "beforeTurn, AskHuman, held messages")
  Rel(toolgw, hitl, "requiresApproval")
  Rel(rep, wsm, "Replica workdir, worktree")
  Rel(eng, brokers, "Declare, Publish, Subscribe, Stats")
```

## Deployment views

### Single host (docker compose)

```mermaid
C4Deployment
  title Deployment — single host

  Deployment_Node(host, "Docker host", "Linux / macOS / Windows") {
    Deployment_Node(net, "compose network weft_default") {
      Container(weftd, "weftd", "weft/weftd", "port 8080")
      Container(agents, "agents", "weft/agents", "read-only FS, cap_drop ALL")
      ContainerQueue(nats, "nats", "nats:2.11 -js")
      ContainerDb(pg, "postgres", "postgres:17")
      Container(obs, "otel-collector, prometheus, tempo, loki, grafana", "observability")
      Container(ollama, "ollama / tgi", "optional profiles")
    }
    Deployment_Node(sbxnet, "compose network weft_sandbox") {
      Container(sbx, "sandbox-daemon", "Weft DinD image + journal, privileged", "TLS :2376 (not published); firewall: sandboxes cannot reach private ranges")
      Container(loader, "sandbox-image", "docker:28.5.2-cli", "one-shot: builds weft/sandbox inside sandbox-daemon")
    }
    Deployment_Node(vols, "Named volumes") {
      ContainerDb(v, "workspaces, weftd-data, nats-data, pg-data, prom/tempo/loki-data, models, sandbox-daemon-data, sandbox-certs-*")
    }
  }
  Rel(weftd, nats, "NATS")
  Rel(weftd, pg, "SQL")
  Rel(weftd, agents, "HTTP")
  Rel(weftd, ollama, "HTTP")
  Rel(weftd, sbx, "Docker API", "mutual TLS")
```

### Highly available (Kubernetes or VMs)

```mermaid
C4Deployment
  title Deployment — highly available

  Deployment_Node(cluster, "Kubernetes cluster / VM fleet") {
    Deployment_Node(cp, "weftd Deployment (N pods)") {
      Container(f1, "weftd-a", "WEFT_INTERNAL_URL=http://<pod-ip>:8080")
      Container(f2, "weftd-b", "same store and broker; leases split replicas")
    }
    Deployment_Node(rt, "Agent runtime Deployments") {
      Container(a1, "weft-rt-<graph>-<node>", "rendered by `weft k8s render`; Service endpoint")
    }
    Deployment_Node(data, "Stateful services") {
      ContainerQueue(js, "NATS JetStream cluster", "R3 streams")
      ContainerDb(pgha, "PostgreSQL HA", "primary + replicas")
      ContainerDb(rwx, "RWX volume", "workspaces shared by weftd pods")
    }
  }
  Rel(f1, js, "")
  Rel(f2, js, "")
  Rel(f1, pgha, "")
  Rel(f2, pgha, "")
  Rel(f1, a1, "POST /v1/turn")
  Rel(a1, f1, "callbacks to the owning pod")
```

In the HA view:

- Workspaces must be on storage that every `weftd` pod can mount (ReadWriteMany), because a replica can move to another instance after failover.
- Callbacks must reach the instance that owns the execution. This is why `WEFT_INTERNAL_URL` is set per pod and does not point at a Service.
